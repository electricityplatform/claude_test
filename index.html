// Edge Function: profile-photo // رفع / تغيير / حذف الصورة الشخصية. بتشتغل بـ service role، فبتعدّي سياسة الحظر على باكيت profile_photos. // // النشر: supabase functions deploy profile-photo --no-verify-jwt // (--no-verify-jwt لازم: التطبيق بيستخدم publishable key مش JWT، والتحقق بيتم جوه الدالة نفسها) // // الطلبات (POST, JSON): // { action: 'upload', user_id, image_base64 } → { ok: true, url } // { action: 'delete', user_id } → { ok: true } import { createClient } from 'npm:@supabase/supabase-js@2'; const BUCKET = 'profile_photos'; const MAX_BYTES = 1024 * 1024; // 1MB (العميل بيصغّر الصورة 512x512 قبل الرفع، فحجمها عادةً أقل من 100KB) const ALLOWED_ORIGINS = ['https://electricityplatform.github.io']; const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i; function corsHeaders(req: Request) { const origin = req.headers.get('origin') ?? ''; return { 'Access-Control-Allow-Origin': ALLOWED_ORIGINS.includes(origin) ? origin : ALLOWED_ORIGINS[0], 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Vary': 'Origin', }; } function ascii(b: Uint8Array, from: number, to: number) { return String.fromCharCode(...b.slice(from, to)); } // بنحدد نوع الصورة من محتواها الفعلي (مش من اللي العميل بيقوله) function detectImage(b: Uint8Array): { type: string; ext: string } | null { if (b.length > 3 && b[0] === 0xff && b[1] === 0xd8 && b[2] === 0xff) return { type: 'image/jpeg', ext: 'jpg' }; if (b.length > 8 && b[0] === 0x89 && b[1] === 0x50 && b[2] === 0x4e && b[3] === 0x47) return { type: 'image/png', ext: 'png' }; if (b.length > 12 && ascii(b, 0, 4) === 'RIFF' && ascii(b, 8, 12) === 'WEBP') return { type: 'image/webp', ext: 'webp' }; return null; } // مسار الملف جوه الباكيت من الرابط المحفوظ في users.profile_photo_url function pathFromUrl(url: string | null): string | null { if (!url) return null; const marker = `/${BUCKET}/`; const i = url.indexOf(marker); if (i === -1) return null; try { return decodeURIComponent(url.slice(i + marker.length).split('?')[0]); } catch { return null; } } Deno.serve(async (req: Request) => { const cors = corsHeaders(req); const reply = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...cors, 'Content-Type': 'application/json' } }); if (req.method === 'OPTIONS') return new Response('ok', { headers: cors }); if (req.method !== 'POST') return reply({ ok: false, error: 'طريقة الطلب غير مدعومة' }, 405); let body: any; try { body = await req.json(); } catch { return reply({ ok: false, error: 'طلب غير صالح' }, 400); } const action = body?.action; const userId = String(body?.user_id ?? ''); if (!UUID_RE.test(userId)) return reply({ ok: false, error: 'معرّف المستخدم غير صالح' }, 400); if (action !== 'upload' && action !== 'delete') return reply({ ok: false, error: 'إجراء غير معروف' }, 400); const supabase = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!, { auth: { persistSession: false }, }); const { data: user, error: userErr } = await supabase .from('users') .select('id, profile_photo_url') .eq('id', userId) .maybeSingle(); if (userErr) return reply({ ok: false, error: 'تعذّر قراءة بيانات المستخدم' }, 500); if (!user) return reply({ ok: false, error: 'المستخدم غير موجود' }, 404); const storage = supabase.storage.from(BUCKET); const oldPath = pathFromUrl(user.profile_photo_url); // يمسح كل ملفات المستخدم القديمة (ماعدا keepName لو اتحدد) + الملف المسجّل في الرابط القديم async function removeOld(keepPath: string | null) { const toRemove = new Set(); const { data: files } = await storage.list(userId, { limit: 100 }); for (const f of files ?? []) { const p = `${userId}/${f.name}`; if (p !== keepPath) toRemove.add(p); } if (oldPath && oldPath !== keepPath) toRemove.add(oldPath); if (toRemove.size) await storage.remove([...toRemove]); } if (action === 'delete') { const { error } = await supabase.from('users').update({ profile_photo_url: null }).eq('id', userId); if (error) return reply({ ok: false, error: 'تعذّر حذف الصورة' }, 500); await removeOld(null); return reply({ ok: true }); } // ---- upload ---- const b64 = String(body?.image_base64 ?? ''); if (!b64 || b64.length > MAX_BYTES * 1.4) return reply({ ok: false, error: 'حجم الصورة غير مناسب' }, 413); let bytes: Uint8Array; try { bytes = Uint8Array.from(atob(b64), (c) => c.charCodeAt(0)); } catch { return reply({ ok: false, error: 'بيانات الصورة غير صالحة' }, 400); } if (!bytes.length || bytes.length > MAX_BYTES) return reply({ ok: false, error: 'حجم الصورة غير مناسب' }, 413); const img = detectImage(bytes); if (!img) return reply({ ok: false, error: 'نوع الصورة غير مدعوم (JPG / PNG / WebP)' }, 400); const newPath = `${userId}/${Date.now()}.${img.ext}`; const up = await storage.upload(newPath, bytes, { contentType: img.type, upsert: false, cacheControl: '31536000' }); if (up.error) return reply({ ok: false, error: 'تعذّر رفع الصورة' }, 500); const { data: pub } = storage.getPublicUrl(newPath); const { error: updErr } = await supabase.from('users').update({ profile_photo_url: pub.publicUrl }).eq('id', userId); if (updErr) { await storage.remove([newPath]); return reply({ ok: false, error: 'تعذّر حفظ رابط الصورة' }, 500); } // الصورة الجديدة اتحفظت: نمسح القديمة try { await removeOld(newPath); } catch (e) { console.error('cleanup error', e); } return reply({ ok: true, url: pub.publicUrl }); });
